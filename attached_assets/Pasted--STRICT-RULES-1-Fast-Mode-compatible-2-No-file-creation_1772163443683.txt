MOCK i18n SETUP:
- Create a temporary mock i18n function in `client/src/i18n.ts`:
  
  ```ts
  import translations from "./locales/ar/translation.json";
  export function t(key: string) {
    const keys = key.split(".");
    let value: any = translations;
    for (const k of keys) {
      if (value[k] !== undefined) value = value[k];
      else return key;
    }
    return value;
  }
